% arXiv version of critical-values.tex.
%
% arXiv processes submissions with LaTeX/pdfLaTeX and does not run XeTeX, so
% this file replaces the fontspec/unicode-math setup of the original with a
% pdfLaTeX one and carries no system fonts. Palatino text with Euler math keeps
% the original's colour; the body below is otherwise identical.
%
% The bibliography is inlined as thebibliography rather than left to
% biblatex+biber, so no .bbl has to be shipped or version-matched.
%
% Build: pdflatex critical-values-arxiv (twice, for the cross-references).
\documentclass[10pt]{article}

\usepackage[T1]{fontenc}
\usepackage[utf8]{inputenc}
\usepackage{newpxtext}    % Palatino-like text, in place of Miller Display
\usepackage{amsmath}
\usepackage{newpxmath}    % Palatino-matched italic math
% Not eulervm: the original loads it but unicode-math overrides it, so the
% XeLaTeX build actually sets math in Latin Modern Math -- italic letters.
% Euler here would set them upright and look nothing like the original.

\usepackage{geometry}
\geometry{letterpaper, margin=1in}
\renewcommand{\baselinestretch}{1.2}

\usepackage{titlesec}
\titleformat{\section}{\normalfont\large\bfseries}{\thesection}{1em}{}
\titleformat{\subsection}{\normalfont\normalsize\bfseries}{\thesubsection}{1em}{}

\usepackage{url}
\usepackage{hyperref}

  pdflang=en-US,

  citecolor = blue}
