\begin{center}
  {\Large \textsc Every Algebraic Number is a
    Critical Value of an Integer Polynomial}

  {by RJ Acu\~na}
\end{center}

\begin{center}
  \begin{minipage}{0.85\textwidth}
    \begin{abstract}
      \noindent
      \justifying
      For every nonconstant $f \in \Z[X]$ we construct $g \in \Z[X]$ and an
      irreducible $H \in \Z[X]$ with $H \mid g'$ and $H^{2} \mid f \circ g$.
      Equivalently, every algebraic number is a critical value of an integer
      polynomial, so the set of critical values of $\Z[X]$ is exactly
      $\overline{\Q}$. The construction is explicit. The proof was
      assisted by AI, and has been formalized in Lean.
    \end{abstract}
  \end{minipage}
\end{center}

\section{Introduction}

A critical value of a nonconstant $g \in \Z[X]$ is a number $\alpha$ with
$g(\beta) = \alpha$ and $g'(\beta) = 0$ for some $\beta$. Since $\overline{\Q}$ is
closed under polynomial evaluation, every critical value is algebraic. This
note proves the converse, and does so with an explicit construction.

\begin{theorem}\label{T1}
  Let $f \in \Z[X]$ be nonconstant. Then there exist $g, H \in \Z[X]$ with $H$
  irreducible in $\Z[X]$ and $1 \leq \deg H$, such that
  \[
    H \mid g' \quad\text{and}\quad H^{2} \mid f \circ g \qquad \text{in }\Z[X].
  \]
  The $g$ produced satisfies $\deg g \geq 2$.
\end{theorem}

The corollary is the statement in the title.

\begin{cor}\label{C3}
  The set of critical values is exactly $\overline{\Q}$.
\end{cor}

\section{The construction}\label{Scons}

Let $h \in \Z[X]$ be a primitive irreducible factor of $f$, of degree $n$, with
root $\theta$ and $h(0) \neq 0$. Put
\[
  \rho = \operatorname{Res}(h,h') \in \Z, \qquad
  v \in \Z[X], \quad \deg v < n, \quad v\,h' \equiv \rho \pmod{h},
\]
which the extended Euclidean algorithm produces, and set
\[
  M = \rho\, h(0), \qquad
  H(X) = \frac{h(MX)}{h(0)}, \qquad
  W(X) = \frac{v(0)H(X) - v(MX)}{\rho} .
\]
The polynomial claimed by Theorem \ref{T1} is
\[
  \boxed{\; g(X) = MX + h(MX)\,W(X) \;}
\]
together with this $H$.

\begin{theorem*}[restatement of Theorem \ref{T1}]
  $H$ is irreducible in $\Z[X]$, and $H \mid g'$ and $H^{2} \mid f \circ g$ in
  $\Z[X]$.
\end{theorem*}

\begin{center}
\fbox{%
  \begin{minipage}{0.93\textwidth}
    \textbf{Verification.} Let $\beta = \theta/M$, a root of $H$. Since
    $H(0) = 1$, the constant terms in $W$ cancel, and every other coefficient
    carries $M^{j}$ with $\rho\, h(0) \mid M$, so $W \in \Z[X]$ and $W(0) = 0$;
    likewise $H, g \in \Z[X]$. Then $W(\beta) = -v(\theta)/\rho = -1/h'(\theta)$,
    so
    \[
      g(\beta) = \theta, \qquad
      g'(\beta) = M\bigl(1 + h'(\theta)W(\beta)\bigr) = 0 .
    \]
    Hence $\beta$ is a double root of $f \circ g$, and $H$ is primitive
    irreducible, so $H \mid g'$ and $H^{2} \mid f(g)$ in $\Z[X]$ by Gauss.
  \end{minipage}}
\end{center}

The computation uses $h(\theta) = 0$ and nothing else about $\theta$, so it runs
at every root of $h$ at once: all $n$ conjugates are critical values of the same
$g$, at the $n$ points $\theta_{i}/M$. That is what makes $H^{2}$, and not
merely $(X - \beta)^{2}$, divide $f \circ g$.

Only the congruence $v h' \equiv \rho \pmod h$ is used, so any nonzero $\rho$
admitting one will serve; the resultant is simply the standard choice. Smaller
choices give a smaller $g$, and the best available is the generator of
$(h,h') \cap \Z$ --- see Example \ref{Ex82}, where $\rho = 2$ while
$\operatorname{Res}(h,h') = 4$.

\section{Examples}\label{Sex}

Each line below is a closed identity, checkable by expansion.

\begin{example}[$f = qX - p$, $p \neq 0$]
  With $u = 0$, $v = 1$, $\rho = q$, $h_{0} = -p$, $M = -pq$, $H = q^{2}X + 1$
  and $W = qX$,
  \[
    g = -pq^{3}X^{2} - 2pqX, \qquad
    g' = -2pq\,(q^{2}X+1), \qquad
    f \circ g = -p\,(q^{2}X+1)^{2}.
  \]
\end{example}

\begin{example}[$f = X^{2}+1$]\label{Ex82}
  With $u = 2$, $v = -X$, $\rho = 2$, $h_{0} = 1$, $M = 2$, $H = 4X^{2}+1$ and
  $W = X$,
  \[
    g = 4X^{3} + 3X, \qquad
    g' = 3(4X^{2}+1), \qquad
    f \circ g = (4X^{2}+1)^{2}(X^{2}+1).
  \]
  Here $\deg g = 3 = n + 1$, the least degree any solution can have. Note
  also $\rho = 2$ while $\operatorname{Res}(h,h') = 4$: the reduced resultant is
  the better choice. The unit group $\mathcal{O}_{K}^{\times} = \{\pm 1, \pm i\}$
  has rank $0$, and it is $\Z[i/2] = \Z[i][1/2]$ that supplies the needed unit
  $1+i$.
\end{example}

\begin{example}[$f = X^{3}-2$]
  With $u = 3$, $v = -X$, $\rho = -6$, $h_{0} = -2$, $M = 12$,
  $H = 1 - 864X^{3}$ and $W = -2X$,
  \[
    g = -3456X^{4} + 16X, \qquad g' = 16\,(1 - 864X^{3}).
  \]
\end{example}

\begin{example}[$f = X^{3}-X-1$]
  The construction gives $M = 23$ and a sextic. A hand-tuned quintic also
  witnesses the \emph{statement}, and is a useful independent check of it:
  \[
    g = -24334X^{5} + 39675X^{4} - 26450X^{3} + 9085X^{2} - 1610X + 118,
    \qquad H = 23X^{3} - 23X^{2} + 8X - 1,
  \]
  with $g' = -230\,(23X-7)H$ and $H^{2} \mid f \circ g$, the cofactor having
  degree $9$.
\end{example}

\begin{example}[Dedekind's cubic]\label{Ex85}
  $f = X^{3} - X^{2} - 2X - 8$, the standard example of a field whose ring of
  integers is not monogenic: $2$ is a common index divisor
  \cite{dedekind1878}. With $u = 21X - 125$, $v = -7X^{2} + 44X - 3$,
  $\rho = 1006$, $h_{0} = -8$, $M = -8048$,
  \[
    H = 65158925824X^{3} + 8096288X^{2} - 2012X + 1, \qquad
    W = -194310912X^{3} + 426544X^{2} + 358X,
  \]
  \[
    \begin{aligned}
      g = {}& 101288722414414331904X^{6} - 209759614012620800X^{5} \\
            & {} - 217370176548864X^{4} - 14767629312X^{3}
                 + 2350016X^{2} - 10912X .
    \end{aligned}
  \]
\end{example}

\section{Formalization}\label{Sformal}

The development above has been formalized in Lean~4 \cite{lean4} against
Mathlib \cite{mathlib}, in about $1{,}200$ lines. The main statement is

\begin{lstlisting}[language=,caption={}]
theorem main (f : Z[X]) (hf : 1 <= f.natDegree) :
    exists g H : Z[X], Irreducible H /\ H | derivative g /\ H^2 | f.comp g
\end{lstlisting}

\noindent
with a strengthened form carrying $2 \leq \deg g$ and $1 \leq \deg H$, and with
Corollary \ref{C3} proved from it. Every result depends only on the three
standard axioms \texttt{propext}, \texttt{Classical.choice} and
\texttt{Quot.sound}.
