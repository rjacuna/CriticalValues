    Critical Value of an Integer Polynomial}

  {by RJ Acu\~na}
\end{center}

\begin{center}
  \begin{minipage}{0.85\textwidth}
    \begin{abstract}
      \noindent
      \justifying
      For every irreducible $f \in \Z[x]$ of positive degree we construct
      $g \in \Z[x]$ and an irreducible $F \in \Z[x]$ with $F \mid g'$ and
      $F^{2} \mid f \circ g$.
      Equivalently, every algebraic number is a critical value of an integer
      polynomial, so the set of critical values of $\Z[x]$ is exactly
      $\overline{\Q}$. The construction is explicit. The proof was
      assisted by AI, and has been formalized in Lean.
    \end{abstract}
  \end{minipage}
\end{center}

\section{Introduction}

A critical value of a nonconstant $g \in \Z[x]$ is a number $\alpha$ with
$g(\beta) = \alpha$ and $g'(\beta) = 0$ for some $\beta$. Since $\overline{\Q}$ is
closed under polynomial evaluation, every critical value is algebraic.
Johannes Walcher posed the problem of establishing the reverse implication.
This note solves Walcher's problem with an explicit construction.

\begin{theorem}\label{T1}
  Let $f$ be an irreducible integral polynomial of positive degree. Then there
  exist $F$ and $g$ in $\Z[x]$, with $F$ irreducible and of positive degree,
  such that
  \[
    F \mid g' \quad\text{and}\quad F^{2} \mid f \circ g .
  \]
\end{theorem}

The corollary is the statement in the title.

\begin{cor}\label{C3}
  The set of critical values is exactly $\overline{\Q}$.
\end{cor}

\begin{pf}
  Write $f = m_{\alpha}$. If $g(\beta) = \alpha$ and $g'(\beta) = 0$ then
  $\beta$ is a double root of $f \circ g$, since $(f \circ g)' = f'(g)\,g'$; so
  $m_{\beta} \mid g'$ and $m_{\beta}^{2} \mid f \circ g$. Conversely take $F$
  and $g$ from Theorem \ref{T1}: a root $\beta$ of $F$ has $g'(\beta) = 0$ and
  $f(g(\beta)) = 0$, so $g(\beta)$ is a conjugate of $\alpha$, and conjugation
  gives $\alpha$ itself.
\end{pf}

\section{The construction}\label{Scons}

Let $f$ be irreducible of degree $n \geq 1$ and $\alpha$ a root. Write
$f(x) = \sum_{k=0}^{n} a_{k}x^{n-k}$, so that $a_{n} = f(0)$. If $f = x$, then
$g = x^{2}$ and $F = x$ work. Otherwise $f(0) \neq 0$, and since
$\gcd(f,f') = 1$ in $\Q[x]$, the elimination ideal $(f,f') \cap \Z$ is nonzero;
let $u, v \in \Z[x]$ and $\rho \in \Z$, $\rho \neq 0$, be such that
$u\,f + v\,f' = \rho$, and set $M := \rho\,a_{n}$, $F(x) := f(Mx)/a_{n}$ and
$W(x) := \bigl(v(0)F(x) - v(Mx)\bigr)/\rho$. The polynomial claimed by
Theorem \ref{T1} is
\begin{equation}\label{Eg}
  g(x) := Mx + f(Mx)\,W(x).
\end{equation}

\begin{theorem*}[restatement of Theorem \ref{T1}]
  $F$ is irreducible, and $F \mid g'$ and $F^{2} \mid f \circ g$.
\end{theorem*}

\begin{pf}
  Write $F(x) = \sum_{k=0}^{n} A_{k}x^{n-k}$. From $A_{k} = a_{k}M^{n-k}/a_{n}$
  we get $A_{n} = 1$ and
  $A_{k} = a_{k}\rho^{\,n-k}a_{n}^{\,n-k-1}$ for $k < n$, so $F \in \Z[x]$ with
  $F(0) = 1$, which gives $W(0) = 0$. And $\rho$ divides $M$ and every $A_{k}$
  with $k < n$, so it divides $v(0)F(x) - v(Mx)$ and $W \in \Z[x]$. Therefore
  $g$ is an integral polynomial.

  Let $\beta = \alpha/M$, a root of $F$; therefore $g(\beta) = \alpha$, and
  $\rho W(\beta) = -v(\alpha)$. Then
  \begin{align*}
    g'(\alpha/M)
      &= M + \rho\,a_{n}f'(\alpha)\,W(\alpha/M)
         + \cancel{f(\alpha)}\,W'(\alpha/M) \\
      &= M - a_{n}f'(\alpha)v(\alpha) = 0 ,
  \end{align*}
  because $u(\alpha)\cancel{f(\alpha)} + v(\alpha)f'(\alpha) = \rho$.
  Hence $\beta$ is a double root of $f \circ g$, and $F$ is primitive
  irreducible, so $F \mid g'$ and $F^{2} \mid f \circ g$ by Gauss.
\end{pf}

Any $\rho$ will do, but $M = \rho\,a_{n}$ multiplies into every coefficient of
$F$, $W$ and $g$, so the reduced resultant gives the answer with the smallest
coefficients.

\section{Examples}\label{Sex}

Each line below is a closed identity, checkable by expansion.

\begin{example}[$f = qx - p$, $p \neq 0$, $\gcd(p,q) = 1$]
  Here $f' = q$ and $p = qx - f$, so $(f,f') \cap \Z = \Z$ and $\rho = 1$.
  Choose $a, b \in \Z$ with $ap + bq = 1$ and take $u = -a$, $v = ax + b$; then
  $f(0) = -p$, $M = -p$, $F = qx + 1$, $W = x$, and
  \[
    g = -pqx^{2} - 2px, \qquad
    g' = -2p\,(qx+1), \qquad
    f \circ g = -p\,(qx+1)^{2}.
  \]
\end{example}

\begin{example}[$f = ax^{2} + bx + c$]\label{Exquad}
  This is irreducible of degree $2$ exactly when $a \neq 0$, $\gcd(a,b,c) = 1$,
  and $D := b^{2} - 4ac$ is not a square. Since $(f')^{2} = 4af + D$,
  \[
    \rho = \operatorname{Res}(f,f') = -aD, \qquad u = 4a^{2}, \qquad v = -af',
  \]
  so $M = -acD$ and
  \[
    F = a^{3}cD^{2}x^{2} - abD\,x + 1, \qquad
    W = a^{3}bcD\,x^{2} - a(b^{2}-2ac)\,x ,
  \]
  \[
    g = a^{6}bc^{3}D^{3}x^{4} - 2a^{4}c^{2}(b^{2}-ac)D^{2}x^{3}
        + a^{2}bc(b^{2}-ac)D\,x^{2} - 2ac(b^{2}-3ac)\,x ,
  \]
  with $g' = 2ac\bigl(2a^{2}bcD\,x - (b^{2}-3ac)\bigr)F$. For $b = 0$ the
  quartic term drops and $\deg g = 3 = n+1$, the least any solution can have.

\end{example}

\begin{example}[$f = x^{3}-x-1$]
  The construction gives $M = 23$ and a sextic. A hand-tuned quintic also
  witnesses the \emph{statement}, and is a useful independent check of it:
  \[
    g = -24334x^{5} + 39675x^{4} - 26450x^{3} + 9085x^{2} - 1610x + 118,
    \qquad F = 23x^{3} - 23x^{2} + 8x - 1,
  \]
  with $g' = -230\,(23x-7)F$ and $F^{2} \mid f \circ g$, the cofactor having
  degree $9$.
\end{example}

\begin{example}[Dedekind's cubic]\label{Ex85}
  $f = x^{3} - x^{2} - 2x - 8$, the standard example of a field whose ring of
  integers is not monogenic: $2$ is a common index divisor
  \cite{dedekind1878}. With $u = 21x - 125$, $v = -7x^{2} + 44x - 3$,
  $\rho = 1006$, $f(0) = -8$, $M = -8048$,
  \[
    F = 65158925824x^{3} + 8096288x^{2} - 2012x + 1, \qquad
    W = -194310912x^{3} + 426544x^{2} + 358x,
  \]
  \[
    \begin{aligned}
      g = {}& 101288722414414331904x^{6} - 209759614012620800x^{5} \\
            & {} - 217370176548864x^{4} - 14767629312x^{3}
                 + 2350016x^{2} - 10912x .
    \end{aligned}
  \]
\end{example}

\section{Formalization}\label{Sformal}

The development above has been formalized in Lean~4 \cite{lean4} against
Mathlib \cite{mathlib}, in about $1{,}200$ lines. The main statement is

\begin{lstlisting}[language=,caption={}]
theorem main (f : Z[X]) (hf : 1 <= f.natDegree) :
    exists g H : Z[X], Irreducible H /\ H | derivative g /\ H^2 | f.comp g
\end{lstlisting}

\noindent
with a strengthened form carrying $2 \leq \deg g$ and $1 \leq \deg H$, and with
Corollary \ref{C3} proved from it. Every result depends only on the three
standard axioms \texttt{propext}, \texttt{Classical.choice} and
\texttt{Quot.sound}.
